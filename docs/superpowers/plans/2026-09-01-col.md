# Col Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implementar Col (19.º juego): juego combinatorio partidista de
colorear regiones de un mapa sin tocar tu propio color; pierde quien no
puede colorear.

**Architecture:** Motor puro sin DOM (`engine.ts`) siguiendo el patrón de
`obstruccion`/`nim`: `createInitialState` / `playMove` que devuelve el
mismo objeto ante jugadas inválidas y nunca muta. Los 3 mapas son datos
estáticos (`maps.ts`). `Board.astro` pinta un SVG con un `<path>` por
región, gestiona un selector de mapa previo y se conecta al arnés
remoto compartido `iniciarSesionJuego`.

**Tech Stack:** Astro (componentes `.astro` con `<script>` TS), Vitest,
TypeScript. SVG inline para el tablero. Sin dependencias nuevas.

**Spec:** `docs/superpowers/specs/2026-09-01-col-design.md`

## Nota de fidelidad al spec

El spec dice "3 mapas SVG dibujados a mano con regiones tipo países".
Este plan concreta esa decisión como **mosaicos rectangulares autorados
a mano** (cada región es un rectángulo con `d` trivial `M x y H x2 V y2
H x1 Z`). Motivo: elimina errores de geometría y de adyacencia, se ve
bien para un juego de colorear infantil, y las adyacencias siguen
escritas a mano en los datos. Si se prefieren siluetas irregulares, se
puede reemplazar `path` y `adyacencias` de cada mapa sin tocar el motor
ni el Board.

## Global Constraints

- **Idioma:** todo el texto de cara al usuario en español. Nombres de
  símbolos por jugador: `●` (Jugador 1), `▲` (Jugador 2).
- **TypeScript:** el repo raíz está pineado a TS `^6` (memoria
  `dependabot-typescript-pin`). No introducir sintaxis que requiera TS 7.
- **Motor puro:** `engine.ts` no importa nada del DOM ni de Astro.
  `playMove` devuelve **el mismo objeto** `state` ante cualquier jugada
  inválida y **nunca** muta su entrada.
- **Modo remoto:** `jugar()` en el Board **no** debe contener ninguna
  comprobación `esMiTurno` (memoria `remoto-guard-turno-jugar`). El
  bloqueo de input vive solo en `render()` y en el selector.
- **Colores:** usar las variables CSS globales existentes
  `--color-player-1` y `--color-player-2`. No hardcodear hex de jugador.
- **Comandos:** usar `rtk` como prefijo (`rtk npm test`, `rtk git ...`).
- **Tests:** `npm test` corre Vitest sobre todos los `*.test.ts`.

function esConexo(mapa: ColMap): boolean {
  const visitadas = new Set<number>([0]);
  const cola = [0];
  while (cola.length) {
    const r = cola.pop()!;
    for (const v of adyacentesDe(mapa, r)) {
      if (!visitadas.has(v)) {
        visitadas.add(v);
        cola.push(v);
      }
    }
  }

}

export function sonAdyacentes(mapa: ColMap, a: number, b: number): boolean {
  if (a === b) return false;
  const lo = Math.min(a, b);
  const hi = Math.max(a, b);
  return mapa.adyacencias.some(([x, y]) => x === lo && y === hi);

export function adyacentesDe(mapa: ColMap, region: number): number[] {
  const res: number[] = [];
  for (const [a, b] of mapa.adyacencias) {
    if (a === region) res.push(b);
    else if (b === region) res.push(a);
  }
  return res;
}

```ts
import { MAPAS, adyacentesDe } from './maps';

export type Player = 1 | 2;
export type RegionColor = 0 | 1 | 2; // 0 = sin color
export type Fase = 'seleccion' | 'jugando' | 'terminado';

export type ColMove =
  | { tipo: 'mapa'; mapaId: number }
  | { tipo: 'color'; region: number };

export interface ColState {
  mapaId: number | null;
  fase: Fase;
  colores: RegionColor[];
  currentPlayer: Player;
  jugadorInicial: Player;
  status: 'playing' | 'won';
  winner: Player | null;
  lastMove: number | null;
}

export function createInitialState(jugadorInicial: Player = 1): ColState {
  return {
    mapaId: null,
    fase: 'seleccion',
    colores: [],
    currentPlayer: jugadorInicial,
    jugadorInicial,
    status: 'playing',
    winner: null,
    lastMove: null,
  };
}

const otro = (p: Player): Player => (p === 1 ? 2 : 1);

export function esJugadaValida(payload: unknown): payload is ColMove {
  if (typeof payload !== 'object' || payload === null || Array.isArray(payload)) {
    return false;
  }
  const p = payload as Record<string, unknown>;
  if (p.tipo === 'mapa') {
    return (
      typeof p.mapaId === 'number' &&
      Number.isInteger(p.mapaId) &&
      p.mapaId >= 0 &&
      p.mapaId < MAPAS.length
    );
  }
  if (p.tipo === 'color') {
    return typeof p.region === 'number' && Number.isInteger(p.region) && p.region >= 0;
  }
  return false;
}

export function regionesLegales(state: ColState, player: Player): number[] {
  if (state.fase !== 'jugando' || state.mapaId === null) return [];
  const mapa = MAPAS[state.mapaId];
  const res: number[] = [];
  for (const region of mapa.regiones) {
    if (state.colores[region.id] !== 0) continue;
    const tocaMiColor = adyacentesDe(mapa, region.id).some(
      v => state.colores[v] === player
    );
    if (!tocaMiColor) res.push(region.id);
  }
  return res;
}

// playMove se añade en la Task 3.
export { otro };
```

- Produces:
  - `export function playMove(state: ColState, move: ColMove): ColState`
    - jugada inválida → devuelve **el mismo objeto** `state`.
    - `{ tipo: 'mapa', mapaId }` en `fase: 'seleccion'` → `fase: 'jugando'`,
      `colores` = ceros de longitud `MAPAS[mapaId].regiones.length`,
      `mapaId` fijado, **turno sin cambiar**, `lastMove: null`.
    - `{ tipo: 'color', region }` en `fase: 'jugando'` y region legal para
      `currentPlayer` → colorea; si el rival queda sin `regionesLegales`
      → `status: 'won'`, `winner: currentPlayer`, `fase: 'terminado'`;
      si no → turno al otro jugador.

describe('playMove — fase jugando', () => {
  function trasElegir(mapaId = 0, inicial: 1 | 2 = 1): ColState {
    return playMove(createInitialState(inicial), { tipo: 'mapa', mapaId });
  }

  it('colorear una región vacía sin vecinas de mi color alterna el turno', () => {
    const s = trasElegir(0);
    const r = playMove(s, { tipo: 'color', region: 0 });
    expect(r.colores[0]).toBe(1);
    expect(r.currentPlayer).toBe(2);
    expect(r.lastMove).toBe(0);
    expect(r.status).toBe('playing');
  });

  it('colorear una región ya coloreada no hace nada', () => {
    let s = trasElegir(0);
    s = playMove(s, { tipo: 'color', region: 0 }); // J1
    expect(playMove(s, { tipo: 'color', region: 0 })).toBe(s); // J2 sobre la misma
  });

  it('J1 no puede colorear una región vecina a otra suya', () => {
    let s = trasElegir(0);
    s = playMove(s, { tipo: 'color', region: 0 }); // J1 -> region 0
    s = playMove(s, { tipo: 'color', region: 2 }); // J2 -> region 2 (no adyacente a 0)
    // turno de J1; region 1 es adyacente a 0 (color de J1) -> ilegal
    expect(playMove(s, { tipo: 'color', region: 1 })).toBe(s);
  });

  it('detecta la victoria cuando el rival se queda sin jugadas', () => {
    // Mapa 0: la región 10 (esquina) sólo es adyacente a 7 y 11.

```ts
export function playMove(state: ColState, move: ColMove): ColState {
  if (state.status !== 'playing') return state;
  if (!esJugadaValida(move)) return state;

  if (move.tipo === 'mapa') {
    if (state.fase !== 'seleccion') return state;
    const mapa = MAPAS[move.mapaId];
    if (!mapa) return state;
    return {
      ...state,
      mapaId: move.mapaId,
      fase: 'jugando',
      colores: Array(mapa.regiones.length).fill(0) as RegionColor[],
      lastMove: null,
    };
  }

  // move.tipo === 'color'
  if (state.fase !== 'jugando' || state.mapaId === null) return state;
  const mapa = MAPAS[state.mapaId];
  if (move.region >= mapa.regiones.length) return state;
  if (!regionesLegales(state, state.currentPlayer).includes(move.region)) {
    return state;
  }

  const colores = [...state.colores];
  colores[move.region] = state.currentPlayer;
  const siguiente: ColState = {
    ...state,
    colores,
    lastMove: move.region,
  };

  const rival = otro(state.currentPlayer);
  if (regionesLegales(siguiente, rival).length === 0) {
    return {
      ...siguiente,
      status: 'won',
      winner: state.currentPlayer,
      fase: 'terminado',
    };
  }

  return { ...siguiente, currentPlayer: rival };
}
```

```astro
<script>
  import {
    createInitialState,
    esJugadaValida,
    playMove,
    regionesLegales,
    type ColState,
    type ColMove,
    type Player,
  } from './engine';
  import { MAPAS } from './maps';
  import { iniciarSesionJuego } from '../../lib/gameSession';

  const selectorEl = document.getElementById('col-selector')!;
  const esperaEl = document.getElementById('col-espera')!;
  const tableroEl = document.getElementById('col-tablero')!;
  const svg = document.getElementById('col-svg') as unknown as SVGSVGElement;
  const btnAleatorio = document.getElementById('col-aleatorio') as HTMLButtonElement;
  const botonesMapa = Array.from(
    selectorEl.querySelectorAll<HTMLButtonElement>('.col-selector__mapa')
  );

  let state: ColState = createInitialState(1);
  let jugadorInicialSiguiente: Player = 1;
  let pathsPorId: SVGPathElement[] = [];

  const sesion = iniciarSesionJuego<ColMove>({
    validarMovimiento: esJugadaValida,
    onMovimientoRemoto: move => jugar(move, false),
    onAplicarReinicio: () => {
      state = createInitialState(jugadorInicialSiguiente);
      pathsPorId = [];
      render();
    },
    onRender: render,
    onDesconectar: () => {
      botonesMapa.forEach(b => (b.disabled = true));
      btnAleatorio.disabled = true;
      pathsPorId.forEach(p => (p.style.pointerEvents = 'none'));
    },
  });

  function construirSvgDelMapa(mapaId: number): void {
    const mapa = MAPAS[mapaId];
    svg.setAttribute('viewBox', mapa.viewBox);
    // conservar <defs>, borrar paths previos
    pathsPorId.forEach(p => p.remove());
    pathsPorId = mapa.regiones.map(r => {
      const p = document.createElementNS('http://www.w3.org/2000/svg', 'path');
      p.setAttribute('d', r.path);
      p.setAttribute('class', 'col-region');
      p.dataset.region = String(r.id);
      p.addEventListener('click', () => {
        jugar({ tipo: 'color', region: r.id });
      });
      svg.appendChild(p);
      return p;
    });
  }

  function render(): void {
    if (state.fase === 'seleccion') {
      tableroEl.hidden = true;
      const meTocaElegir = sesion.esMiTurno(state.currentPlayer);
      selectorEl.hidden = !meTocaElegir;
      esperaEl.hidden = meTocaElegir;
      if (!meTocaElegir) {
        esperaEl.textContent = `Esperando a que ${
          sesion.nombres[state.jugadorInicial]
        } elija el mapa…`;
      }
      return;
    }

    selectorEl.hidden = true;
    esperaEl.hidden = true;
    tableroEl.hidden = false;

    if (state.mapaId !== null && pathsPorId.length !== MAPAS[state.mapaId].regiones.length) {
      construirSvgDelMapa(state.mapaId);
    }

    const esMiTurno = sesion.esMiTurno(state.currentPlayer);
    const legales =
      state.fase === 'jugando' && sesion.miAsiento
        ? new Set(regionesLegales(state, sesion.miAsiento))
        : new Set<number>();

    pathsPorId.forEach((p, id) => {
      const color = state.colores[id];
      p.classList.toggle('col-region--j1', color === 1);
      p.classList.toggle('col-region--j2', color === 2);
      const legal = esMiTurno && legales.has(id);
      p.classList.toggle('col-region--legal', legal);
      p.classList.toggle('col-region--ultima', state.lastMove === id);
      p.style.pointerEvents = legal ? 'auto' : 'none';
    });

    if (state.status === 'playing') {
      sesion.mostrarTurno({
        jugador: state.currentPlayer,
        simbolos: { 1: '●', 2: '▲' },
      });
    } else if (state.status === 'won') {
      const simbolo = state.winner === 1 ? '●' : '▲';
      sesion.mostrarFinDeJuego({
        titulo: `🎉 ¡Ganó ${sesion.nombres[state.winner!]} (${simbolo})! 🖍️`,
      });
    }
  }

  function jugar(move: ColMove, emitirRemoto = true): void {
    const nuevo = playMove(state, move);
    if (nuevo === state) return;
    state = nuevo;
    if (nuevo.status === 'won' && nuevo.winner) {
      jugadorInicialSiguiente = nuevo.winner === 1 ? 2 : 1;
    }
    render();
    if (emitirRemoto) sesion.enviarMovimiento(move);
  }

  botonesMapa.forEach(boton => {
    boton.addEventListener('click', () => {
      jugar({ tipo: 'mapa', mapaId: Number(boton.dataset.mapa) });
    });
  });

  btnAleatorio.addEventListener('click', () => {
    jugar({ tipo: 'mapa', mapaId: Math.floor(Math.random() * MAPAS.length) });
  });

  render();
</script>
```
