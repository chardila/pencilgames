```ts
export type Player = 1 | 2;
export type CellValue = Player | null;
export type GameStatus = 'playing' | 'won';
export type BoardSize = 7 | 9 | 11;

export interface YState {
  size: BoardSize;
  board: CellValue[];
  currentPlayer: Player;
  status: GameStatus;
  winner: Player | null;
  winningCells: number[] | null;
  lastMove: number | null;
}

// 6 direcciones sobre el tablero triangular: [dr, dc]
const Y_DIRECTIONS: ReadonlyArray<readonly [number, number]> = [
  [0, -1],
  [0, 1],
  [-1, -1],
  [-1, 0],
  [1, 0],
  [1, 1],
];

export function cellCount(size: BoardSize): number {
  return (size * (size + 1)) / 2;
}

export function indexOf(r: number, c: number): number {
  return (r * (r + 1)) / 2 + c;
}

export function coordsOf(index: number): { r: number; c: number } {
  const r = Math.floor((Math.sqrt(8 * index + 1) - 1) / 2);
  const c = index - (r * (r + 1)) / 2;
  return { r, c };
}

export function createInitialState(size: BoardSize = 9): YState {
  return {
    size,
    board: Array<CellValue>(cellCount(size)).fill(null),
    currentPlayer: 1,
    status: 'playing',
    winner: null,
    winningCells: null,
    lastMove: null,
  };
}

export function esJugadaValida(
  payload: unknown,
  size: BoardSize
): payload is number {
  return (
    typeof payload === 'number' &&
    Number.isInteger(payload) &&
    payload >= 0 &&
    payload < cellCount(size)
  );
}

export function getNeighbors(index: number, size: BoardSize): number[] {
  const { r, c } = coordsOf(index);
  const neighbors: number[] = [];
  for (const [dr, dc] of Y_DIRECTIONS) {
    const nr = r + dr;
    const nc = c + dc;
    if (nr >= 0 && nr < size && nc >= 0 && nc <= nr) {
      neighbors.push(indexOf(nr, nc));
    }
  }
  return neighbors;
}
```

```ts
export function tocaLados(
  cells: number[],
  size: BoardSize
): { izq: boolean; der: boolean; inf: boolean } {
  let izq = false;
  let der = false;
  let inf = false;
  for (const i of cells) {
    const { r, c } = coordsOf(i);
    if (c === 0) izq = true;
    if (c === r) der = true;
    if (r === size - 1) inf = true;
  }
  return { izq, der, inf };
}

export function playMove(state: YState, index: number): YState {
  if (state.status !== 'playing') return state;
  if (!esJugadaValida(index, state.size)) return state;
  if (state.board[index] !== null) return state;

  const player = state.currentPlayer;
  const board = [...state.board];
  board[index] = player;

  // Flood-fill BFS desde la ficha recién colocada, solo por celdas de `player`.
  const componente: number[] = [];
  const visitado = new Set<number>([index]);
  const cola: number[] = [index];
  while (cola.length > 0) {
    const actual = cola.shift()!;
    componente.push(actual);
    for (const vecino of getNeighbors(actual, state.size)) {
      if (board[vecino] === player && !visitado.has(vecino)) {
        visitado.add(vecino);
        cola.push(vecino);
      }
    }
  }

  const { izq, der, inf } = tocaLados(componente, state.size);
  if (izq && der && inf) {
    return {
      ...state,
      board,
      status: 'won',
      winner: player,
      winningCells: componente,
      lastMove: index,
    };
  }

  return {
    ...state,
    board,
    currentPlayer: player === 1 ? 2 : 1,
    lastMove: index,
  };
}
```

```ts
// Verificación independiente: ¿algún componente conectado de `player`
// toca los tres lados?
function ganadorPorFloodFill(
  board: (1 | 2 | null)[],
  size: BoardSize
): 1 | 2 | null {
  for (const player of [1, 2] as const) {
    const visitado = new Set<number>();
    for (let inicio = 0; inicio < cellCount(size); inicio++) {
      if (board[inicio] !== player || visitado.has(inicio)) continue;
      const comp: number[] = [];
      const cola = [inicio];
      visitado.add(inicio);
      while (cola.length > 0) {
        const a = cola.shift()!;
        comp.push(a);
        for (const v of getNeighbors(a, size)) {
          if (board[v] === player && !visitado.has(v)) {
            visitado.add(v);
            cola.push(v);
          }
        }
      }
      const { izq, der, inf } = tocaLados(comp, size);
      if (izq && der && inf) return player;
    }
  }
  return null;
}

function barajar<T>(arr: T[]): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

describe('propiedad: Y no tiene empates', () => {
  for (const N of [7, 9, 11] as BoardSize[]) {
    it(`N=${N}: toda partida aleatoria termina con exactamente un ganador`, () => {
      const iteraciones = 5000;
      for (let it = 0; it < iteraciones; it++) {
        let s = createInitialState(N);
        const orden = barajar([...Array(cellCount(N)).keys()]);
        for (const celda of orden) {
          if (s.status === 'won') break;
          s = playMove(s, celda);
        }
        // Con el tablero lleno (o antes) siempre hay ganador.
        expect(s.status).toBe('won');
        expect(s.winner === 1 || s.winner === 2).toBe(true);
        // El ganador del motor coincide con la verificación independiente.
        expect(ganadorPorFloodFill(s.board, N)).toBe(s.winner);
      }
    });
  }
});
```

```astro
    indexOf,

    type YState,
  } from './engine';

  const SVG_NS = 'http://www.w3.org/2000/svg';
  const svg = document.getElementById('y-svg') as unknown as SVGSVGElement;

  const R = 24; // radio del hexágono
  const W = Math.sqrt(3) * R;
  const MARGIN = 42;

  function hexCenter(r: number, c: number, size: number): { cx: number; cy: number } {
    // Fila r desplazada a la izquierda media celda por nivel para formar el
    // triángulo equilátero con ápice arriba.
    const offsetX = MARGIN + ((size - 1) / 2) * W + R;
    const cx = offsetX + (c - r / 2) * W;
    const cy = MARGIN + r * (1.5 * R) + R;
    return { cx, cy };
  }

  function hexCorners(cx: number, cy: number): string {
    const points: string[] = [];
    for (let k = 0; k < 6; k++) {
      const angle = ((30 + 60 * k) * Math.PI) / 180;
      points.push(
        `${(cx + R * Math.cos(angle)).toFixed(1)},${(cy + R * Math.sin(angle)).toFixed(1)}`
      );
    }
    return points.join(' ');
  }

  function construirTablero(): void {
    svg.innerHTML = '';
    const size = state.size;

    const totalWidth = MARGIN * 2 + size * W;
    const totalHeight = MARGIN * 2 + (size - 1) * (1.5 * R) + 2 * R;
    svg.setAttribute('viewBox', `0 0 ${totalWidth.toFixed(1)} ${totalHeight.toFixed(1)}`);

    // Bandas de borde neutras (los tres lados son de ambos jugadores).
    const gBordes = document.createElementNS(SVG_NS, 'g');
    const lado = (pts: string[]) => {
      const pl = document.createElementNS(SVG_NS, 'polyline');
      pl.setAttribute('points', pts.join(' '));
      pl.setAttribute('class', 'y-borde-banda');
      gBordes.appendChild(pl);
    };
    const v = (cx: number, cy: number, gradoAngulo: number) => {
      const a = (gradoAngulo * Math.PI) / 180;
      return `${(cx + R * Math.cos(a)).toFixed(1)},${(cy + R * Math.sin(a)).toFixed(1)}`;
    };

    // Lado izquierdo: celdas (r, 0), vértices exteriores izquierdos.
    const ptsIzq: string[] = [];
    for (let r = 0; r < size; r++) {
      const { cx, cy } = hexCenter(r, 0, size);
      if (r === 0) ptsIzq.push(v(cx, cy, 270));
      ptsIzq.push(v(cx, cy, 210), v(cx, cy, 150));

    lado(ptsIzq);

    // Lado derecho: celdas (r, r), vértices exteriores derechos.

      const { cx, cy } = hexCenter(r, r, size);
      if (r === 0) ptsDer.push(v(cx, cy, 270));
      ptsDer.push(v(cx, cy, 330), v(cx, cy, 30));

    lado(ptsDer);

      const { cx, cy } = hexCenter(size - 1, c, size);
      if (c === 0) ptsInf.push(v(cx, cy, 150));
      ptsInf.push(v(cx, cy, 90), v(cx, cy, 30));
    }
    lado(ptsInf);

    svg.appendChild(gBordes);

    // Celdas.
    const gCeldas = document.createElementNS(SVG_NS, 'g');
    for (let i = 0; i < cellCount(size); i++) {
      const { r, c } = coordsOf(i);
      const { cx, cy } = hexCenter(r, c, size);

      const g = document.createElementNS(SVG_NS, 'g');
      g.setAttribute('class', 'y-grupo-celda');
      g.setAttribute('data-indice', String(i));
      g.setAttribute('tabindex', '0');
      g.setAttribute('role', 'button');
      g.setAttribute('aria-label', `Fila ${r + 1}, celda ${c + 1}`);

      const poly = document.createElementNS(SVG_NS, 'polygon');
      poly.setAttribute('points', hexCorners(cx, cy));
      poly.setAttribute('class', 'y-celda');
      poly.setAttribute('data-indice', String(i));

      const text = document.createElementNS(SVG_NS, 'text');
      text.setAttribute('x', cx.toFixed(1));
      text.setAttribute('y', (cy + 1).toFixed(1));
      text.setAttribute('class', 'y-glifo');

      g.appendChild(poly);
      g.appendChild(text);
      g.addEventListener('click', () => jugar(i));
      g.addEventListener('keydown', (e: KeyboardEvent) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          jugar(i);
        }
      });
      gCeldas.appendChild(g);
    }
    svg.appendChild(gCeldas);
  }

  function jugar(indice: number, emitirRemoto = true): void {
    if (state.board[indice] !== null || state.status !== 'playing') return;
    if (emitirRemoto && !sesion.esMiTurno(state.currentPlayer)) return;

    state = playMove(state, indice);
    render();
    if (emitirRemoto) {
      sesion.enviarMovimiento(indice);
    }
  }

  function render(): void {
    const jugadorDelTurno = state.currentPlayer;
    const esMiTurno = sesion.esMiTurno(jugadorDelTurno);

    const hayJugadas = state.board.some(cel => cel !== null);
    botonesTamano.forEach(btn => {
      const btnSize = Number(btn.dataset.tamano);
      const isActivo = btnSize === state.size;
      btn.classList.toggle('btn-tamano--activo', isActivo);
      btn.setAttribute('aria-checked', isActivo ? 'true' : 'false');
      btn.disabled = hayJugadas || sesion.miAsiento !== null;
    });

    const grupos = svg.querySelectorAll<SVGGElement>('.y-grupo-celda');
    grupos.forEach((grupo, i) => {
      const valor = state.board[i];
      const poly = grupo.querySelector<SVGPolygonElement>('.y-celda')!;
      const text = grupo.querySelector<SVGTextElement>('.y-glifo')!;
      const { r, c } = coordsOf(i);

      if (valor) {
        poly.dataset.ocupada = 'true';
        poly.dataset.jugador = String(valor);
        const simbolo = valor === 1 ? '●' : '▲';
        text.textContent = simbolo;
        grupo.setAttribute('aria-label', `Fila ${r + 1}, celda ${c + 1} (Jugador ${valor} ${simbolo})`);
        grupo.setAttribute('aria-disabled', 'true');
      } else {
        delete poly.dataset.ocupada;
        delete poly.dataset.jugador;
        text.textContent = '';
        grupo.setAttribute('aria-label', `Fila ${r + 1}, celda ${c + 1}`);
        if (state.status !== 'playing' || !esMiTurno) {
          grupo.setAttribute('aria-disabled', 'true');
        } else {
          grupo.removeAttribute('aria-disabled');
        }
      }

      poly.classList.toggle('y-celda--ultima', state.lastMove === i);
      poly.classList.toggle(
        'y-celda--ganadora',
        state.winningCells?.includes(i) ?? false
      );

      const interactiva = valor === null && state.status === 'playing' && esMiTurno;
      poly.style.pointerEvents = interactiva ? 'auto' : valor ? 'default' : 'none';
    });

    if (state.status === 'playing') {
      sesion.mostrarTurno({
        jugador: jugadorDelTurno,
        simbolos: { 1: '●', 2: '▲' },
      });
    } else if (state.status === 'won') {
      const ganador = state.winner!;
      const simbolo = ganador === 1 ? '●' : '▲';
      sesion.mostrarFinDeJuego({
        titulo: `🎉 ¡Ganó ${sesion.nombres[ganador]} (${simbolo})!`,
        detalle: 'Conectó los tres lados del triángulo',
      });
    }
  }

  botonesTamano.forEach(btn => {
    btn.addEventListener('click', () => {
      if (state.board.some(cel => cel !== null) || sesion.miAsiento !== null) return;
      currentSize = Number(btn.dataset.tamano) as BoardSize;
      state = createInitialState(currentSize);
      construirTablero();
      render();
    });
  });

  construirTablero();
  render();
</script>
```

```markdown
# Y

## Players
2

## Goal
Connect all three sides of a triangular board using one continuous chain of your stones.

## Setup
Use a triangular board of hexagonal cells. Side length N has N(N+1)/2 cells
(NOT N×N). Recommended sizes: N = 7, 9, or 11 (28, 45, 66 cells).

Each player is assigned one color.

## Turn
Place one stone of your color on any empty cell.

## Victory
A player wins when a single connected chain of their stones touches all three
sides of the triangle at once. The three sides are goals for BOTH players
(unlike Hex, where each player owns one pair of edges).

## Indexing & connectivity
Row `r` (0..N-1, apex to base) holds cells `c` in 0..r; `index = r*(r+1)/2 + c`.
Neighbors of `(r,c)`: `(r,c±1)`, `(r-1,c-1)`, `(r-1,c)`, `(r+1,c)`, `(r+1,c+1)`,
each bounds-checked as `0 <= r' <= N-1` and `0 <= c' <= r'`.
Sides: left = `c === 0`, right = `c === r`, bottom = `r === N-1`.

## Implementation
On each placement, flood-fill from the just-placed stone only, across same-color
neighbors. Track which of the three sides that one component touches; the mover
wins iff it touches all three. Do NOT seed the search from every cell of one
side at once — that merges disjoint components and reports false wins.

## Note
Y is closely related to Hex but has a three-sided connection objective and never
draws (a full board always has exactly one winner).
```
