**Goal:** Añadir Nim como decimoctavo juego de Pencilgames: dos jugadores retiran estrellas por turnos de una pirámide 1-3-5-7; gana quien toma la última.

**Architecture:** Motor puro sin DOM (`engine.ts`) con estado inmutable y funciones `createInitialState` / `esJugadaValida` / `playMove`, probado con Vitest (TDD). **Tech Stack:** Astro 7, TypeScript, Vitest, `<TableroJuego>` + `gameSession.ts` (WebRTC P2P para modo remoto).

**Spec:** `docs/superpowers/specs/2026-09-01-nim-design.md`

- Node `>=22.12.0` (ver `package.json` `engines`).

```ts
export type Player = 1 | 2;
export type GameStatus = 'playing' | 'won';

export const FILAS_INICIALES = [1, 3, 5, 7] as const;
export const TOTAL_FICHAS = 16;

export interface NimMove {
  fila: number;
  dejar: number;
}

export interface NimState {
  montones: number[];
  currentPlayer: Player;
  status: GameStatus;
  winner: Player | null;
  lastMove: { fila: number; quitadas: number } | null;
}

export function createInitialState(): NimState {
  return {
    montones: [...FILAS_INICIALES],
    currentPlayer: 1,
    status: 'playing',
    winner: null,
    lastMove: null,
  };
}

export function esJugadaValida(payload: unknown): payload is NimMove {
  if (typeof payload !== 'object' || payload === null || Array.isArray(payload)) {
    return false;
  }
  const { fila, dejar } = payload as Record<string, unknown>;
  return (
    typeof fila === 'number' &&
    Number.isInteger(fila) &&
    fila >= 0 &&
    fila <= 3 &&
    typeof dejar === 'number' &&
    Number.isInteger(dejar) &&
    dejar >= 0
  );
}

export function playMove(state: NimState, move: NimMove): NimState {
  return state;
}
```

Reemplazar el cuerpo de `playMove` en `engine.ts`:

```ts
export function playMove(state: NimState, move: NimMove): NimState {
  if (state.status !== 'playing') return state;
  if (!esJugadaValida(move)) return state;

  const actual = state.montones[move.fila];
  if (move.dejar >= actual) return state; // hay que retirar al menos una

  const montones = [...state.montones];
  montones[move.fila] = move.dejar;
  const quitadas = actual - move.dejar;
  const lastMove = { fila: move.fila, quitadas };

  if (montones.every(n => n === 0)) {
    return {
      montones,
      currentPlayer: state.currentPlayer,
      status: 'won',
      winner: state.currentPlayer,
      lastMove,
    };
  }

  return {
    montones,
    currentPlayer: state.currentPlayer === 1 ? 2 : 1,
    status: 'playing',
    winner: null,
    lastMove,
  };
}
```

```astro
    const esMiTurno = sesion.esMiTurno(jugadorDelTurno);

    estrellas.forEach(e => {
      const { fila, pos } = filaPos(e);
      const quitada = pos >= state.montones[fila];
      const esUltima =
        !!state.lastMove &&
        state.lastMove.fila === fila &&
        pos >= state.montones[fila] &&
        pos < state.montones[fila] + state.lastMove.quitadas;
      const enSeleccion =
        !!seleccion && seleccion.fila === fila && pos >= seleccion.dejar && !quitada;

      e.classList.toggle('estrella--quitada', quitada);
      e.classList.toggle('estrella--ultima', esUltima && quitada);
      e.classList.toggle('estrella--seleccionada', enSeleccion);
      e.disabled = quitada || state.status !== 'playing' || !esMiTurno;
    });

    if (seleccion && state.status === 'playing' && esMiTurno) {
      const quitadas = state.montones[seleccion.fila] - seleccion.dejar;
      barraTexto.textContent = `Quitar ${quitadas} ${
        quitadas === 1 ? 'estrella' : 'estrellas'
      } de la fila ${seleccion.fila + 1}`;
      barra.hidden = false;
    } else {
      barra.hidden = true;
    }

    if (state.status === 'playing') {
      sesion.mostrarTurno({ jugador: jugadorDelTurno });
    } else if (state.status === 'won') {
      sesion.mostrarFinDeJuego({
        titulo: `🎉 ¡Ganó ${sesion.nombres[state.winner!]}! ⭐`,
      });
    }
  }

  function jugar(move: NimMove, emitirRemoto = true): void {
    state = playMove(state, move);
    seleccion = null;
    limpiarPreview();
    render();
    if (emitirRemoto) {
      sesion.enviarMovimiento(move);
    }
  }

  estrellas.forEach(estrella => {
    estrella.addEventListener('click', () => {
      const { fila, pos } = filaPos(estrella);
      if (seleccion && seleccion.fila === fila && seleccion.dejar === pos) {
        jugar({ fila, dejar: pos });
        return;
      }
      seleccion = { fila, dejar: pos };
      limpiarPreview();
      render();
    });

    estrella.addEventListener('pointerenter', () => {
      if (estrella.disabled) return;
      const { fila, pos } = filaPos(estrella);
      limpiarPreview();
      estrellas.forEach(e => {
        const fp = filaPos(e);
        if (fp.fila === fila && fp.pos >= pos && fp.pos < state.montones[fila]) {
          e.classList.add('estrella--preview');
        }
      });
    });
  });

  tablero.addEventListener('pointerleave', limpiarPreview);

  btnConfirmar.addEventListener('click', () => {
    if (seleccion) jugar({ fila: seleccion.fila, dejar: seleccion.dejar });
  });

  btnCancelar.addEventListener('click', () => {
    seleccion = null;
    limpiarPreview();
    render();
  });

  render();
</script>
```
