**Goal:** Añadir Chomp (barra de chocolate 4×7 con onza envenenada) como octavo juego del sitio, jugable en modo local y en modo remoto.

**Architecture:** Motor puro (`src/games/chomp/engine.ts`) sin DOM, testeado exhaustivamente con Vitest (TDD), junto con un componente `Board.astro` que recrea visualmente una tableta de chocolate y conecta las interacciones mediante `iniciarSesionJuego`. Toda la infraestructura compartida (indicador de turnos, banner de fin de juego, identidad de jugadores y canal WebRTC P2P) se reutiliza sin modificaciones.

**Tech Stack:** Astro, TypeScript estricto, CSS nativo fluido, Vitest, npm.

**Spec:** `docs/superpowers/specs/2026-08-30-chomp-design.md`

## Global Constraints

- Tablero **4 filas × 7 columnas** (28 casillas, índices `0` a `27` en orden fila-mayor).
- Casilla `0` (Fila 0, Columna 0): es la onza envenenada (`☠️`). Está deshabilitada durante la partida para evitar clics accidentales.
- Victoria: el jugador que realiza un movimiento que deja **únicamente** la casilla `0` en el tablero gana de inmediato (`status: 'won'`, `winner: currentPlayer`).
- Mordisco: elegir una casilla `(r, c)` elimina esa casilla y todas las casillas `(i, j)` con `i >= r` y `j >= c`.
- Payload de movimiento remoto: índice `number` en `[1, 27]`.
- Motor inmutable: ante entrada inválida o jugada ilegal, `playMove` devuelve el `state` recibido sin mutarlo.
- Solo español en todo el texto visible.
- `engine.ts` no importa nada del DOM ni de Astro.
- Registro estático: `import` estático de `Board.astro` en `src/pages/juegos/[slug].astro`.
- TDD: escribir los tests primero, verificar que fallan, e implementar el mínimo código necesario para que pasen.

```ts
// src/games/chomp/engine.ts
export type Player = 1 | 2;
export type GameStatus = 'playing' | 'won';

export interface ChompState {
  board: boolean[];
  currentPlayer: Player;
  status: GameStatus;
  winner: Player | null;
  lastEaten: number[];
  lastMove: number | null;
}

export const FILAS = 4;
export const COLUMNAS = 7;
export const TOTAL_CASILLAS = FILAS * COLUMNAS;
export const INDICE_VENENO = 0;

export function createInitialState(): ChompState {
  return {
    board: Array<boolean>(TOTAL_CASILLAS).fill(true),
    currentPlayer: 1,
    status: 'playing',
    winner: null,
    lastEaten: [],
    lastMove: null,
  };
}

export function esJugadaValida(payload: unknown): payload is number {
  return (
    typeof payload === 'number' &&
    Number.isInteger(payload) &&
    payload >= 1 &&
    payload < TOTAL_CASILLAS
  );
}

export function playMove(state: ChompState, index: number): ChompState {
  if (state.status !== 'playing') return state;
  if (!esJugadaValida(index)) return state;
  if (!state.board[index]) return state;

  const targetRow = Math.floor(index / COLUMNAS);
  const targetCol = index % COLUMNAS;

  const newBoard = [...state.board];
  const eaten: number[] = [];

  for (let i = 0; i < TOTAL_CASILLAS; i++) {
    if (!newBoard[i]) continue;
    const r = Math.floor(i / COLUMNAS);
    const c = i % COLUMNAS;
    if (r >= targetRow && c >= targetCol) {
      newBoard[i] = false;
      eaten.push(i);
    }
  }

  const casillasVivas = newBoard.filter(c => c === true).length;

  if (casillasVivas === 1) {
    return {
      board: newBoard,
      currentPlayer: state.currentPlayer,
      status: 'won',
      winner: state.currentPlayer,
      lastEaten: eaten,
      lastMove: index,
    };
  }

  return {
    board: newBoard,
    currentPlayer: state.currentPlayer === 1 ? 2 : 1,
    status: 'playing',
    winner: null,
    lastEaten: eaten,
    lastMove: index,
  };
}
```

<script>
  import {
    createInitialState,
    esJugadaValida,
    playMove,
    COLUMNAS,
    INDICE_VENENO,
    type ChompState,
  } from './engine';
  import { iniciarSesionJuego } from '../../lib/gameSession';

  const tablero = document.getElementById('tablero')!;
  const casillas = Array.from(
    tablero.querySelectorAll<HTMLButtonElement>('.casilla')
  );

  let state: ChompState = createInitialState();

  const sesion = iniciarSesionJuego<number>({
    validarMovimiento: esJugadaValida,
    onMovimientoRemoto: indice => jugar(indice, false),
    onAplicarReinicio: () => {
      state = createInitialState();
      render();
    },
    onRender: render,
    onDesconectar: () => {
      casillas.forEach(casilla => (casilla.disabled = true));
    },
  });

  function render(): void {
    const jugadorDelTurno = state.currentPlayer;
    const esMiTurno = sesion.esMiTurno(jugadorDelTurno);

    casillas.forEach((casilla, i) => {
      const viva = state.board[i];
      casilla.classList.toggle('casilla--comida', !viva);
      casilla.classList.toggle(
        'casilla--ultima-comida',
        state.lastEaten.includes(i) && !viva
      );

      if (i === INDICE_VENENO) {
        casilla.disabled = true;
      } else {
        casilla.disabled = !viva || state.status !== 'playing' || !esMiTurno;
      }
    });

    if (state.status === 'playing') {
      sesion.mostrarTurno({
        jugador: jugadorDelTurno,
      });
    } else if (state.status === 'won') {
      sesion.mostrarFinDeJuego({
        titulo: `🎉 ¡Ganó ${sesion.nombres[state.winner!]}! 🍫`,
      });
    }
  }

  function jugar(indice: number, emitirRemoto = true): void {
    state = playMove(state, indice);
    render();
    if (emitirRemoto) {
      sesion.enviarMovimiento(indice);
    }
  }

  // Previsualización en hover (área del mordisco)
  function actualizarPreview(indiceHover: number | null): void {
    if (indiceHover === null || indiceHover === INDICE_VENENO || !state.board[indiceHover]) {
      casillas.forEach(c => c.classList.remove('casilla--preview'));
      return;
    }
    const targetRow = Math.floor(indiceHover / COLUMNAS);
    const targetCol = indiceHover % COLUMNAS;

    casillas.forEach((casilla, i) => {
      if (i === INDICE_VENENO || !state.board[i]) {
        casilla.classList.remove('casilla--preview');
        return;
      }
      const r = Math.floor(i / COLUMNAS);
      const c = i % COLUMNAS;
      casilla.classList.toggle(
        'casilla--preview',
        r >= targetRow && c >= targetCol
      );
    });
  }

  casillas.forEach(casilla => {
    casilla.addEventListener('click', () => {
      const idx = Number(casilla.dataset.indice);
      actualizarPreview(null);
      jugar(idx);
    });

    casilla.addEventListener('pointerenter', () => {
      const idx = Number(casilla.dataset.indice);
      actualizarPreview(idx);
    });
  });

  tablero.addEventListener('pointerleave', () => {
    actualizarPreview(null);
  });

  render();
</script>
