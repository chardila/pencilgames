## Global Constraints

- Tablero romboidal con celdas hexagonales *pointy-topped*.
- Tamaños soportados: `5×5`, `7×7` (por defecto) y `9×9`.
- Array plano de longitud $N^2$ con orden fila-mayor ($i = r \cdot N + c$).
- 6 vecinos hexagonales por casilla $(r, c)$: $[-1, 0], [-1, 1], [0, -1], [0, 1], [1, -1], [1, 0]$ acotados estrictamente por $0 \le r < N$ y $0 \le c < N$.
- **Jugador 1** (`●` en `var(--color-player-1)` / Naranja): Conecta Norte ($r = 0$) con Sur ($r = N - 1$).
- **Jugador 2** (`▲` en `var(--color-player-2)` / Azul): Conecta Oeste ($c = 0$) con Este ($c = N - 1$).
- Sin empates. Gana inmediatamente quien forme una cadena ininterrumpida entre sus dos bordes.
- Sin regla de intercambio (*swap*).
- Victoria única por partida (sin marcador de puntos acumulado).
- Payload del movimiento remoto: índice `number` en $[0, N^2 - 1]$.
- Motor inmutable: ante entradas inválidas devuelve el `state` intacto.
- Solo español en todo el texto visible.
- `engine.ts` no importa nada del DOM ni de Astro.
- Registro estático en `src/pages/juegos/[slug].astro`.

```ts
// src/games/hex/engine.ts
export type Player = 1 | 2;
export type CellValue = Player | null;
export type GameStatus = 'playing' | 'won';
export type BoardSize = 5 | 7 | 9;

export interface HexState {
  size: BoardSize;
  board: CellValue[];
  currentPlayer: Player;
  status: GameStatus;
  winner: Player | null;
  winningPath: number[] | null;
  lastMove: number | null;
}

const HEX_DIRECTIONS: ReadonlyArray<readonly [number, number]> = [
  [-1, 0],  // Noroeste
  [-1, 1],  // Noreste
  [0, -1],  // Oeste
  [0, 1],   // Este
  [1, -1],  // Suroeste
  [1, 0],   // Sureste
];

export function createInitialState(size: BoardSize = 7): HexState {
  return {
    size,
    board: Array<CellValue>(size * size).fill(null),
    currentPlayer: 1,
    status: 'playing',
    winner: null,
    winningPath: null,
    lastMove: null,
  };
}

export function esJugadaValida(payload: unknown, size: BoardSize = 7): payload is number {
  return (
    typeof payload === 'number' &&
    Number.isInteger(payload) &&
    payload >= 0 &&
    payload < size * size
  );
}

export function getNeighbors(index: number, size: BoardSize): number[] {
  const r = Math.floor(index / size);
  const c = index % size;
  const neighbors: number[] = [];

  for (const [dr, dc] of HEX_DIRECTIONS) {
    const nr = r + dr;
    const nc = c + dc;
    if (nr >= 0 && nr < size && nc >= 0 && nc < size) {
      neighbors.push(nr * size + nc);
    }
  }

  return neighbors;
}

export function playMove(state: HexState, index: number): HexState {
  if (state.status !== 'playing') return state;
  if (!esJugadaValida(index, state.size)) return state;
  if (state.board[index] !== null) return state;

  const board = [...state.board];
  board[index] = state.currentPlayer;

  return {
    ...state,
    board,
    currentPlayer: state.currentPlayer === 1 ? 2 : 1,
    lastMove: index,
  };
}
```

```ts
export function findWinningPath(
  board: CellValue[],
  size: BoardSize,
  player: Player
): number[] | null {
  const queue: number[] = [];
  const visited = new Set<number>();
  const parentMap = new Map<number, number | null>();

  // Definir nodos iniciales y condición de meta según el jugador
  if (player === 1) {
    // Jugador 1: inicia en fila 0, busca llegar a fila size - 1
    for (let c = 0; c < size; c++) {
      const index = c;
      if (board[index] === 1) {
        queue.push(index);
        visited.add(index);
        parentMap.set(index, null);
      }
    }
  } else {
    // Jugador 2: inicia en col 0, busca llegar a col size - 1
    for (let r = 0; r < size; r++) {
      const index = r * size;
      if (board[index] === 2) {
        queue.push(index);
        visited.add(index);
        parentMap.set(index, null);
      }
    }
  }

  let goalNode: number | null = null;

  while (queue.length > 0) {
    const current = queue.shift()!;
    const r = Math.floor(current / size);
    const c = current % size;

    if (player === 1 && r === size - 1) {
      goalNode = current;
      break;
    }
    if (player === 2 && c === size - 1) {
      goalNode = current;
      break;
    }

    for (const neighbor of getNeighbors(current, size)) {
      if (board[neighbor] === player && !visited.has(neighbor)) {
        visited.add(neighbor);
        parentMap.set(neighbor, current);
        queue.push(neighbor);
      }
    }
  }

  if (goalNode === null) return null;

  // Reconstruir camino desde la meta hasta el inicio
  const path: number[] = [];
  let curr: number | null = goalNode;
  while (curr !== null) {
    path.push(curr);
    curr = parentMap.get(curr) ?? null;
  }

  return path.reverse();
}

export function playMove(state: HexState, index: number): HexState {
  if (state.status !== 'playing') return state;
  if (!esJugadaValida(index, state.size)) return state;
  if (state.board[index] !== null) return state;

  const board = [...state.board];
  board[index] = state.currentPlayer;

  const winningPath = findWinningPath(board, state.size, state.currentPlayer);

  if (winningPath !== null) {
    return {
      ...state,
      board,
      status: 'won',
      winner: state.currentPlayer,
      winningPath,
      lastMove: index,
    };
  }

  return {
    ...state,
    board,
    currentPlayer: state.currentPlayer === 1 ? 2 : 1,
    lastMove: index,
  };
}
```

```astro
<script>
  import {
    createInitialState,
    esJugadaValida,
    playMove,
    type HexState,
    type BoardSize,
  } from './engine';
  import { iniciarSesionJuego } from '../../lib/gameSession';

  const SVG_NS = 'http://www.w3.org/2000/svg';
  const svg = document.getElementById('hex-svg') as unknown as SVGSVGElement;
  const botonesTamano = Array.from(
    document.querySelectorAll<HTMLButtonElement>('.btn-tamano')
  );

  let currentSize: BoardSize = 7;
  let state: HexState = createInitialState(currentSize);

  const sesion = iniciarSesionJuego<number>({
    validarMovimiento: p => esJugadaValida(p, state.size),
    onMovimientoRemoto: idx => jugar(idx, false),
    onAplicarReinicio: () => {
      state = createInitialState(currentSize);
      construirTablero();
      render();
    },
    onRender: render,
    onDesconectar: () => {
      svg.querySelectorAll<SVGPolygonElement>('.hex-celda').forEach(celda => {
        celda.style.pointerEvents = 'none';
      });
      botonesTamano.forEach(btn => (btn.disabled = true));
    },
  });

  const R = 28; // Radio del hexágono
  const W = Math.sqrt(3) * R;
  const MARGIN = 45;

  function hexCenter(r: number, c: number): { cx: number; cy: number } {
    const cx = MARGIN + (c + r * 0.5) * W + R;
    const cy = MARGIN + r * (1.5 * R) + R;
    return { cx, cy };
  }

  function hexCorners(cx: number, cy: number): string {
    const points: string[] = [];
    for (let k = 0; k < 6; k++) {
      const angle = ((30 + 60 * k) * Math.PI) / 180;
      const x = cx + R * Math.cos(angle);
      const y = cy + R * Math.sin(angle);
      points.push(`${x.toFixed(1)},${y.toFixed(1)}`);
    }
    return points.join(' ');
  }

  function construirTablero(): void {
    svg.innerHTML = '';
    const size = state.size;

    const totalWidth = MARGIN * 2 + (size - 1 + (size - 1) * 0.5) * W + 2 * R;
    const totalHeight = MARGIN * 2 + (size - 1) * (1.5 * R) + 2 * R;
    svg.setAttribute('viewBox', `0 0 ${totalWidth.toFixed(1)} ${totalHeight.toFixed(1)}`);

    // Capa de bordes perimetrales de color
    const gBordes = document.createElementNS(SVG_NS, 'g');
    gBordes.setAttribute('class', 'hex-capa-bordes');

    // Borde Norte (J1): conecta centros o vértices superiores de fila 0
    const bordeNorte = document.createElementNS(SVG_NS, 'polyline');
    const ptsNorte: string[] = [];
    for (let c = 0; c < size; c++) {
      const { cx, cy } = hexCenter(0, c);
      const angle1 = (210 * Math.PI) / 180;
      const angle2 = (270 * Math.PI) / 180;
      const angle3 = (330 * Math.PI) / 180;
      if (c === 0) ptsNorte.push(`${(cx + R * Math.cos(angle1)).toFixed(1)},${(cy + R * Math.sin(angle1)).toFixed(1)}`);
      ptsNorte.push(`${(cx + R * Math.cos(angle2)).toFixed(1)},${(cy + R * Math.sin(angle2)).toFixed(1)}`);
      ptsNorte.push(`${(cx + R * Math.cos(angle3)).toFixed(1)},${(cy + R * Math.sin(angle3)).toFixed(1)}`);
    }
    bordeNorte.setAttribute('points', ptsNorte.join(' '));
    bordeNorte.setAttribute('class', 'hex-borde-banda hex-borde-j1');
    gBordes.appendChild(bordeNorte);

    // Borde Sur (J1): conecta vértices inferiores de fila size-1
    const bordeSur = document.createElementNS(SVG_NS, 'polyline');
    const ptsSur: string[] = [];
    for (let c = 0; c < size; c++) {
      const { cx, cy } = hexCenter(size - 1, c);
      const angle1 = (150 * Math.PI) / 180;
      const angle2 = (90 * Math.PI) / 180;
      const angle3 = (30 * Math.PI) / 180;
      if (c === 0) ptsSur.push(`${(cx + R * Math.cos(angle1)).toFixed(1)},${(cy + R * Math.sin(angle1)).toFixed(1)}`);
      ptsSur.push(`${(cx + R * Math.cos(angle2)).toFixed(1)},${(cy + R * Math.sin(angle2)).toFixed(1)}`);
      ptsSur.push(`${(cx + R * Math.cos(angle3)).toFixed(1)},${(cy + R * Math.sin(angle3)).toFixed(1)}`);
    }
    bordeSur.setAttribute('points', ptsSur.join(' '));
    bordeSur.setAttribute('class', 'hex-borde-banda hex-borde-j1');
    gBordes.appendChild(bordeSur);

    // Borde Oeste (J2): conecta vértices izquierdos de col 0
    const bordeOeste = document.createElementNS(SVG_NS, 'polyline');
    const ptsOeste: string[] = [];
    for (let r = 0; r < size; r++) {
      const { cx, cy } = hexCenter(r, 0);
      const angle1 = (270 * Math.PI) / 180;
      const angle2 = (210 * Math.PI) / 180;
      const angle3 = (150 * Math.PI) / 180;
      if (r === 0) ptsOeste.push(`${(cx + R * Math.cos(angle1)).toFixed(1)},${(cy + R * Math.sin(angle1)).toFixed(1)}`);
      ptsOeste.push(`${(cx + R * Math.cos(angle2)).toFixed(1)},${(cy + R * Math.sin(angle2)).toFixed(1)}`);
      ptsOeste.push(`${(cx + R * Math.cos(angle3)).toFixed(1)},${(cy + R * Math.sin(angle3)).toFixed(1)}`);
    }
    bordeOeste.setAttribute('points', ptsOeste.join(' '));
    bordeOeste.setAttribute('class', 'hex-borde-banda hex-borde-j2');
    gBordes.appendChild(bordeOeste);

    // Borde Este (J2): conecta vértices derechos de col size-1
    const bordeEste = document.createElementNS(SVG_NS, 'polyline');
    const ptsEste: string[] = [];
    for (let r = 0; r < size; r++) {
      const { cx, cy } = hexCenter(r, size - 1);
      const angle1 = (270 * Math.PI) / 180;
      const angle2 = (330 * Math.PI) / 180;
      const angle3 = (30 * Math.PI) / 180;
      if (r === 0) ptsEste.push(`${(cx + R * Math.cos(angle1)).toFixed(1)},${(cy + R * Math.sin(angle1)).toFixed(1)}`);
      ptsEste.push(`${(cx + R * Math.cos(angle2)).toFixed(1)},${(cy + R * Math.sin(angle2)).toFixed(1)}`);
      ptsEste.push(`${(cx + R * Math.cos(angle3)).toFixed(1)},${(cy + R * Math.sin(angle3)).toFixed(1)}`);
    }
    bordeEste.setAttribute('points', ptsEste.join(' '));
    bordeEste.setAttribute('class', 'hex-borde-banda hex-borde-j2');
    gBordes.appendChild(bordeEste);

    svg.appendChild(gBordes);

    // Capa de celdas hexagonales
    const gCeldas = document.createElementNS(SVG_NS, 'g');
    gCeldas.setAttribute('class', 'hex-capa-celdas');

    for (let r = 0; r < size; r++) {
      for (let c = 0; c < size; c++) {
        const i = r * size + c;
        const { cx, cy } = hexCenter(r, c);

        const g = document.createElementNS(SVG_NS, 'g');
        g.setAttribute('class', 'hex-grupo-celda');
        g.setAttribute('data-indice', String(i));
        g.setAttribute('tabindex', '0');
        g.setAttribute('role', 'button');
        g.setAttribute('aria-label', `Fila ${r + 1}, columna ${c + 1}`);

        const poly = document.createElementNS(SVG_NS, 'polygon');
        poly.setAttribute('points', hexCorners(cx, cy));
        poly.setAttribute('class', 'hex-celda');
        poly.setAttribute('data-indice', String(i));

        const text = document.createElementNS(SVG_NS, 'text');
        text.setAttribute('x', cx.toFixed(1));
        text.setAttribute('y', (cy + 1).toFixed(1));
        text.setAttribute('class', 'hex-glifo');

        g.appendChild(poly);
        g.appendChild(text);

        g.addEventListener('click', () => jugar(i));
        g.addEventListener('keydown', (e: KeyboardEvent) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            jugar(i);
          }
        });

        gCeldas.appendChild(g);
      }
    }
    svg.appendChild(gCeldas);

    // Capa para la línea de conexión ganadora
    const gVictoria = document.createElementNS(SVG_NS, 'g');
    gVictoria.setAttribute('class', 'hex-capa-victoria');
    svg.appendChild(gVictoria);
  }

  function jugar(indice: number, emitirRemoto = true): void {
    if (state.board[indice] !== null || state.status !== 'playing') return;
    if (!sesion.esMiTurno(state.currentPlayer)) return;

    state = playMove(state, indice);
    render();
    if (emitirRemoto) {
      sesion.enviarMovimiento(indice);
    }
  }

  function render(): void {
    const jugadorDelTurno = state.currentPlayer;
    const esMiTurno = sesion.esMiTurno(jugadorDelTurno);
    const size = state.size;

    // Deshabilitar selector de tamaño una vez que hay jugadas
    const hayJugadas = state.board.some(c => c !== null);
    botonesTamano.forEach(btn => {
      const btnSize = Number(btn.dataset.tamano);
      btn.classList.toggle('btn-tamano--activo', btnSize === size);
      btn.disabled = hayJugadas;
    });

    const grupos = svg.querySelectorAll<SVGGElement>('.hex-grupo-celda');
    grupos.forEach((grupo, i) => {
      const valor = state.board[i];
      const poly = grupo.querySelector<SVGPolygonElement>('.hex-celda')!;
      const text = grupo.querySelector<SVGTextElement>('.hex-glifo')!;

      if (valor) {
        poly.dataset.ocupada = 'true';
        poly.dataset.jugador = String(valor);
        text.textContent = valor === 1 ? '●' : '▲';
      } else {
        delete poly.dataset.ocupada;
        delete poly.dataset.jugador;
        text.textContent = '';
      }

      poly.classList.toggle('hex-celda--ultima', state.lastMove === i);
      const esGanadora = state.winningPath?.includes(i) ?? false;
      poly.classList.toggle('hex-celda--ganadora', esGanadora);

      const interactiva = valor === null && state.status === 'playing' && esMiTurno;
      poly.style.pointerEvents = interactiva ? 'auto' : (valor ? 'default' : 'none');
    });

    // Dibujar línea conectora si hay camino ganador
    const gVictoria = svg.querySelector<SVGGElement>('.hex-capa-victoria');
    if (gVictoria) {
      gVictoria.innerHTML = '';
      if (state.winningPath && state.winningPath.length > 0) {
        const pts = state.winningPath.map(idx => {
          const r = Math.floor(idx / size);
          const c = idx % size;
          const { cx, cy } = hexCenter(r, c);
          return `${cx.toFixed(1)},${cy.toFixed(1)}`;
        }).join(' ');

        const linea = document.createElementNS(SVG_NS, 'polyline');
        linea.setAttribute('points', pts);
        linea.setAttribute('class', 'hex-linea-ganadora');
        gVictoria.appendChild(linea);
      }
    }

    if (state.status === 'playing') {
      sesion.mostrarTurno({
        jugador: jugadorDelTurno,
        simbolos: { 1: '●', 2: '▲' },
      });
    } else if (state.status === 'won') {
      const ganador = state.winner!;
      const simbolo = ganador === 1 ? '●' : '▲';
      sesion.mostrarFinDeJuego({
        titulo: `🎉 ¡Ganó ${sesion.nombres[ganador]} (${simbolo})!`,
        detalle: ganador === 1 ? 'Conexión Norte - Sur completada' : 'Conexión Oeste - Este completada',
      });
    }
  }

  botonesTamano.forEach(btn => {
    btn.addEventListener('click', () => {
      if (state.board.some(c => c !== null)) return;
      currentSize = Number(btn.dataset.tamano) as BoardSize;
      state = createInitialState(currentSize);
      construirTablero();
      render();
    });
  });

  construirTablero();
  render();
</script>
```
