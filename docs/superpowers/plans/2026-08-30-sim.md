## Global Constraints

- Grafo **$K_6$ con 6 vértices** ($0..5$) y **15 aristas canónicas** $(u, v)$ con $0 \le u < v \le 5$.
- Regla misère: el primer jugador en formar un triángulo cuyos 3 lados sean de su propio color **pierde inmediatamente** (`loser = currentPlayer`, `winner = 3 - currentPlayer`, `status = 'finished'`).
- Sin empates: por el Teorema de Ramsey ($R(3,3)=6$), siempre existirá al menos un triángulo monocromático al completar las 15 aristas.
- Payload de movimiento remoto: objeto `Edge` `{ u: Vertex, v: Vertex }`.
- Motor inmutable: ante entradas inválidas o jugadas ilegales, `playMove` devuelve el `state` original sin mutarlo.
- Solo español en todo el texto visible de la interfaz y reglas.
- `engine.ts` no contiene referencias al DOM ni a Astro.
- Registro estático: `import` estático de `Board.astro` en `src/pages/juegos/[slug].astro`.
- TDD estricto: escribir los tests primero, verificar que fallan, e implementar el mínimo código necesario para que pasen.

```ts
// src/games/sim/engine.ts
export type Vertex = 0 | 1 | 2 | 3 | 4 | 5;
export type Player = 1 | 2;
export type SimStatus = 'playing' | 'finished';

export interface Edge {
  u: Vertex;
  v: Vertex;
}

export interface SimState {
  edges: (Player | null)[];
  currentPlayer: Player;
  status: SimStatus;
  winner: Player | null;
  loser: Player | null;
  losingTriangle: [Vertex, Vertex, Vertex] | null;
  moveHistory: Edge[];
}

export const TOTAL_VERTICES = 6;
export const TOTAL_EDGES = 15;

/**
 * Mapeo canónico para un par de vértices (u, v) a un índice 0..14.
 * Para u < v:
 * fila u=0: (0,1)->0, (0,2)->1, (0,3)->2, (0,4)->3, (0,5)->4 (5 aristas)
 * fila u=1: (1,2)->5, (1,3)->6, (1,4)->7, (1,5)->8          (4 aristas)
 * fila u=2: (2,3)->9, (2,4)->10, (2,5)->11                  (3 aristas)
 * fila u=3: (3,4)->12, (3,5)->13                            (2 aristas)
 * fila u=4: (4,5)->14                                       (1 arista)
 */
export function getEdgeIndex(u: number, v: number): number {
  const min = Math.min(u, v);
  const max = Math.max(u, v);
  // Fórmula: sum_{i=0}^{min-1} (TOTAL_VERTICES - 1 - i) + (max - min - 1)
  let index = 0;
  for (let i = 0; i < min; i++) {
    index += TOTAL_VERTICES - 1 - i;
  }
  return index + (max - min - 1);
}

export function getEdgeFromIndex(index: number): Edge {
  let count = 0;
  for (let u = 0; u < TOTAL_VERTICES; u++) {
    const rowCount = TOTAL_VERTICES - 1 - u;
    if (index < count + rowCount) {
      const offset = index - count;
      const v = u + 1 + offset;
      return { u: u as Vertex, v: v as Vertex };
    }
    count += rowCount;
  }
  throw new Error(`Índice de arista fuera de rango: ${index}`);
}

export function getAllEdges(): Edge[] {
  const edges: Edge[] = [];
  for (let i = 0; i < TOTAL_EDGES; i++) {
    edges.push(getEdgeFromIndex(i));
  }
  return edges;
}

export function createInitialState(): SimState {
  return {
    edges: Array<Player | null>(TOTAL_EDGES).fill(null),
    currentPlayer: 1,
    status: 'playing',
    winner: null,
    loser: null,
    losingTriangle: null,
    moveHistory: [],
  };
}

export function esEdgeValido(payload: unknown): payload is Edge {
  if (!payload || typeof payload !== 'object') return false;
  const { u, v } = payload as Record<string, unknown>;
  return (
    typeof u === 'number' &&
    typeof v === 'number' &&
    Number.isInteger(u) &&
    Number.isInteger(v) &&
    u >= 0 &&
    u < TOTAL_VERTICES &&
    v >= 0 &&
    v < TOTAL_VERTICES &&
    u !== v
  );
}
```

```ts
export function playMove(state: SimState, edge: Edge): SimState {
  if (state.status !== 'playing') return state;
  if (!esEdgeValido(edge)) return state;

  const u = Math.min(edge.u, edge.v) as Vertex;
  const v = Math.max(edge.u, edge.v) as Vertex;
  const edgeIndex = getEdgeIndex(u, v);

  if (state.edges[edgeIndex] !== null) return state;

  const player = state.currentPlayer;
  const newEdges = [...state.edges];
  newEdges[edgeIndex] = player;

  // Inspeccionar si se completó un triángulo monocromático con un tercer vértice w
  let fatalTriangle: [Vertex, Vertex, Vertex] | null = null;
  for (let w = 0; w < TOTAL_VERTICES; w++) {
    if (w === u || w === v) continue;
    const edgeUW = newEdges[getEdgeIndex(u, w)];
    const edgeVW = newEdges[getEdgeIndex(v, w)];
    if (edgeUW === player && edgeVW === player) {
      fatalTriangle = [u, v, w as Vertex];
      break;
    }
  }

  const normalizedEdge: Edge = { u, v };

  if (fatalTriangle !== null) {
    return {
      edges: newEdges,
      currentPlayer: player,
      status: 'finished',
      loser: player,
      winner: player === 1 ? 2 : 1,
      losingTriangle: fatalTriangle,
      moveHistory: [...state.moveHistory, normalizedEdge],
    };
  }

  return {
    edges: newEdges,
    currentPlayer: player === 1 ? 2 : 1,
    status: 'playing',
    loser: null,
    winner: null,
    losingTriangle: null,
    moveHistory: [...state.moveHistory, normalizedEdge],
  };
}
```

import TableroJuego from '../../components/TableroJuego.astro';
import { getAllEdges, TOTAL_VERTICES, type Edge, type Vertex } from './engine';

const R = 140;
const CX = 200;
const CY = 200;

// Posiciones angulares de los 6 vértices (empezando arriba a -90°)
const vertices = Array.from({ length: TOTAL_VERTICES }, (_, i) => {
  const angle = -Math.PI / 2 + (i * 2 * Math.PI) / TOTAL_VERTICES;
  const x = CX + R * Math.cos(angle);
  const y = CY + R * Math.sin(angle);
  return { id: i as Vertex, x, y, label: String(i + 1) };
});

const allEdges = getAllEdges().map(edge => {
  const v1 = vertices[edge.u];
  const v2 = vertices[edge.v];
  return {
    ...edge,
    x1: v1.x,
    y1: v1.y,
    x2: v2.x,
    y2: v2.y,
  };

<script>
  import {
    createInitialState,
    getEdgeIndex,
    esEdgeValido,
    playMove,
    TOTAL_VERTICES,
    type SimState,
    type Edge,
    type Vertex,
  } from './engine';
  import { iniciarSesionJuego } from '../../lib/gameSession';

  const svg = document.getElementById('svg-sim') as unknown as SVGSVGElement;
  const trianguloFatal = document.getElementById('triangulo-fatal') as unknown as SVGPolygonElement;
  const aristaPreview = document.getElementById('arista-preview') as unknown as SVGLineElement;
  const hitboxes = Array.from(svg.querySelectorAll<SVGLineElement>('.arista-hitbox'));
  const visuales = Array.from(svg.querySelectorAll<SVGLineElement>('.arista-visual'));
  const nodos = Array.from(svg.querySelectorAll<SVGGElement>('.nodo'));

  const R = 140;
  const CX = 200;
  const CY = 200;
  const vertexCoords: { x: number; y: number }[] = Array.from({ length: TOTAL_VERTICES }, (_, i) => {
    const angle = -Math.PI / 2 + (i * 2 * Math.PI) / TOTAL_VERTICES;
    return {
      x: CX + R * Math.cos(angle),
      y: CY + R * Math.sin(angle),
    };
  });

  let state: SimState = createInitialState();
  let selectedVertex: Vertex | null = null;

  const sesion = iniciarSesionJuego<Edge>({
    validarMovimiento: esEdgeValido,
    onMovimientoRemoto: edge => jugar(edge, false),
    onAplicarReinicio: () => {
      state = createInitialState();
      selectedVertex = null;
      render();
    },
    onRender: render,
    onDesconectar: () => {
      selectedVertex = null;
      actualizarPreview(null);
      render();
    },
  });

  function render(): void {
    const esMiTurno = sesion.esMiTurno(state.currentPlayer) && state.status === 'playing';

    // Renderizado de aristas visuales
    for (const visual of visuales) {
      const u = Number(visual.dataset.u);
      const v = Number(visual.dataset.v);
      const idx = getEdgeIndex(u, v);
      const jugador = state.edges[idx];

      if (jugador) {
        visual.dataset.jugador = String(jugador);
      } else {
        delete visual.dataset.jugador;
      }
    }

    // Hitboxes
    for (const hitbox of hitboxes) {
      const u = Number(hitbox.dataset.u);
      const v = Number(hitbox.dataset.v);
      const idx = getEdgeIndex(u, v);
      const ocupada = state.edges[idx] !== null;

      hitbox.dataset.jugada = String(ocupada);
      if (!esMiTurno || ocupada) {
        hitbox.setAttribute('disabled', 'true');
      } else {
        hitbox.removeAttribute('disabled');
      }
    }

    // Nodos
    for (const nodo of nodos) {
      const idx = Number(nodo.dataset.indice) as Vertex;
      nodo.classList.toggle('seleccionado', selectedVertex === idx);

      let posibleDestino = false;
      if (selectedVertex !== null && selectedVertex !== idx) {
        const edgeIdx = getEdgeIndex(selectedVertex, idx);
        posibleDestino = state.edges[edgeIdx] === null;
      }
      nodo.classList.toggle('posible-destino', posibleDestino);
    }

    // Triángulo fatal
    if (state.status === 'finished' && state.losingTriangle && state.loser) {
      const [v0, v1, v2] = state.losingTriangle;
      const c0 = vertexCoords[v0];
      const c1 = vertexCoords[v1];
      const c2 = vertexCoords[v2];
      trianguloFatal.setAttribute('points', `${c0.x},${c0.y} ${c1.x},${c1.y} ${c2.x},${c2.y}`);
      trianguloFatal.className.baseVal = `triangulo-fatal visible--p${state.loser}`;
    } else {
      trianguloFatal.setAttribute('points', '');
      trianguloFatal.className.baseVal = 'triangulo-fatal';
    }

    // Banner y turnos
    if (state.status === 'playing') {
      sesion.mostrarTurno({
        jugador: state.currentPlayer,
      });
    } else {
      const ganador = state.winner!;
      const perdedor = state.loser!;
      sesion.mostrarFinDeJuego({
        titulo: `🎉 ¡Ganó ${sesion.nombres[ganador]}!`,
        detalle: `${sesion.nombres[perdedor]} cerró un triángulo de su color.`,
      });
    }
  }

  function actualizarPreview(edge: { u: number; v: number } | null): void {
    if (!edge || state.status !== 'playing' || !sesion.esMiTurno(state.currentPlayer)) {
      aristaPreview.setAttribute('x1', '0');
      aristaPreview.setAttribute('y1', '0');
      aristaPreview.setAttribute('x2', '0');
      aristaPreview.setAttribute('y2', '0');
      aristaPreview.className.baseVal = 'arista-preview';
      return;
    }

    const c1 = vertexCoords[edge.u];
    const c2 = vertexCoords[edge.v];
    aristaPreview.setAttribute('x1', String(c1.x));
    aristaPreview.setAttribute('y1', String(c1.y));
    aristaPreview.setAttribute('x2', String(c2.x));
    aristaPreview.setAttribute('y2', String(c2.y));
    aristaPreview.className.baseVal = `arista-preview activa--p${state.currentPlayer}`;
  }

  function jugar(edge: Edge, emitirRemoto = true): void {
    if (state.status !== 'playing') return;
    const prevState = state;
    state = playMove(state, edge);
    if (state === prevState) return;

    selectedVertex = null;
    actualizarPreview(null);
    render();

    if (emitirRemoto) {
      sesion.enviarMovimiento(edge);
    }
  }

  // Interacción en aristas (hitboxes directas)
  for (const hitbox of hitboxes) {
    hitbox.addEventListener('click', () => {
      const u = Number(hitbox.dataset.u) as Vertex;
      const v = Number(hitbox.dataset.v) as Vertex;
      jugar({ u, v });
    });

    hitbox.addEventListener('pointerenter', () => {
      const u = Number(hitbox.dataset.u);
      const v = Number(hitbox.dataset.v);
      const idx = getEdgeIndex(u, v);
      if (state.edges[idx] === null) {
        actualizarPreview({ u, v });
      }
    });
  }

  // Interacción en nodos (flujo nodo-a-nodo)
  for (const nodo of nodos) {
    nodo.addEventListener('click', () => {
      if (state.status !== 'playing' || !sesion.esMiTurno(state.currentPlayer)) return;
      const idx = Number(nodo.dataset.indice) as Vertex;

      if (selectedVertex === null) {
        selectedVertex = idx;
        render();
      } else if (selectedVertex === idx) {
        selectedVertex = null;
        actualizarPreview(null);
        render();
      } else {
        const edgeIdx = getEdgeIndex(selectedVertex, idx);
        if (state.edges[edgeIdx] === null) {
          const edge: Edge = { u: selectedVertex, v: idx };
          jugar(edge);
        } else {
          // Cambia la selección al nodo nuevo si la arista ya estaba ocupada
          selectedVertex = idx;
          actualizarPreview(null);
          render();
        }
      }
    });

    nodo.addEventListener('pointerenter', () => {
      if (selectedVertex !== null) {
        const idx = Number(nodo.dataset.indice) as Vertex;
        if (selectedVertex !== idx) {
          const edgeIdx = getEdgeIndex(selectedVertex, idx);
          if (state.edges[edgeIdx] === null) {
            actualizarPreview({ u: selectedVertex, v: idx });
          }
        }
      }
    });
  }

  svg.addEventListener('pointerleave', () => {
    if (selectedVertex === null) {
      actualizarPreview(null);
    }
  });

  document.addEventListener('keydown', e => {
    if (e.key === 'Escape' && selectedVertex !== null) {
      selectedVertex = null;
      actualizarPreview(null);
      render();
    }
  });

  render();
</script>

title: "Sim"
description: "Traza líneas entre 6 puntos por turnos. ¡Pierde quien primero complete un triángulo de su propio color!"
icono: "🔺"
minJugadores: 2
maxJugadores: 2
---

1. El juego se desarrolla sobre 6 puntos distribuidos en círculo, con 15 líneas posibles para conectar.
2. Cada jugador tiene un color asignado y, por turnos, traza una línea entre dos puntos que aún no estén conectados.
3. **Regla de derrota:** Pierde inmediatamente el jugador que trace una línea cerrando un triángulo cuyos tres lados sean de su propio color.
4. Los triángulos formados por líneas de colores combinados son seguros y no causan derrota.
5. **Dato curioso:** Por el *Teorema de Ramsey* ($R(3,3)=6$), en este tablero es matemáticamente imposible empatar; siempre habrá un perdedor al completarse la partida.
